# migration-safety-known-index-locks.txt
#
# Allowlist for justified exceptions to migration safety rules.
#
# Format (one entry per line):
#   <migration-filename>:<rule-id>
#
# Rule IDs:
#   sensitive-type-change          — ALTER COLUMN … TYPE without USING
#   not-null-without-default       — ADD COLUMN NOT NULL without DEFAULT
#   non-concurrent-index           — CREATE INDEX without CONCURRENTLY
#   constraint-without-not-valid   — ADD CONSTRAINT without NOT VALID
#   rename-column                  — RENAME COLUMN
#   rename-table                   — RENAME TO
#
# Examples (uncomment and edit to use):
#   20250216000000_init.sql:non-concurrent-index          # initial schema — table is empty at creation time
#   20260222000000_transaction_search_indexes.sql:non-concurrent-index  # added during low-traffic window
