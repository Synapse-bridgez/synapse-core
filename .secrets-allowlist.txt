# Secret scanner allowlist. Keep this minimal and explicitly document false positives.
# Entries are regexes matched against repo-relative paths.
#
# NOTE: Broad directory-wide entries (e.g. ^docs/.*$, ^scripts/.*$) are intentionally
# avoided so that real adapter credentials cannot be hidden by placing them in those
# directories. Only narrowly-scoped, explicitly documented false positives are allowed.
#
# --- Documentation ---
# Docs may contain illustrative (non-functional) credential snippets. These are
# documentation examples only and must never be real secrets.
^docs/.*\.md$
^docs/.*\.mdx$
#
# --- Monitoring ---
# Monitoring configs reference secret *names* / env var keys, not values.
^monitoring/.*\.(ya?ml|json)$
#
# --- Test fixtures (non-conformance) ---
# Static fixtures used by unit/integration tests. Fake values only.
^tests/fixtures/.*$
#
# --- Conformance kit fixtures ---
# The conformance suite ships deliberately fake credentials and a deliberately
# non-conformant adapter to prove the suite fails it. These are test-only values
# and are never loaded by production wiring.
^tests/conformance/fixtures/.*$
^tests/conformance/mock_adapter/.*$
#
# --- Certification reports ---
# Machine-readable certification reports are generated artifacts. They must not
# contain secrets (the kit redacts them), but the report schema includes
# credential *fingerprints* (hashes) which trip the scanner. Only the generated
# report output directory is allowlisted, not the certification tooling itself.
^tests/conformance/reports/.*\.json$
^target/certification/.*\.json$
#
# --- Explicitly named fake/example/fixture files anywhere ---
# Narrow filename patterns for obviously-fake credential material.
.*example.*
.*fixture.*
.*fake.*credential.*
.*\.fake\..*
#
# --- Example env files ---
# Template env files with placeholder values only.
^\.env\.example$
^.*\.env\.example$
